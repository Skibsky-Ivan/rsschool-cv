04 / Code

Code example

Climbing stairs task solved with dynamic programming. 

var climbStairs = function(n) {
    if (n < 3) return n;

    let prev1 = 1;
    let prev2 = 2;

    for (let i = 3; i <= n; i++) {
        const current = prev1 + prev2;
        prev1 = prev2;
        prev2 = current;
    }

    return prev2;
};
